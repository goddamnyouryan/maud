       ```bash
       python3 - <<'PY'
       import os, re, sys
       root = os.path.abspath(".maud")
       link_re = re.compile(r'\[[^\]]*\]\(([^)]+)\)')
       broken = []
       checked = 0
       for dp, _, files in os.walk(root):
           for fn in files:
               if not fn.endswith(".md"):
                   continue
               fpath = os.path.join(dp, fn)
               with open(fpath) as f:
                   text = f.read()
               for m in link_re.finditer(text):
                   target = m.group(1).split("#", 1)[0].split(" ", 1)[0]
                   if not target:
                       continue
                   if target.startswith(("http://", "https://", "mailto:")):
                       continue
                   resolved = os.path.normpath(os.path.join(dp, target))
                   checked += 1
                   if target.endswith("/"):
                       # Directory link — must be a real directory containing README.md
                       ok = os.path.isdir(resolved) and os.path.isfile(os.path.join(resolved, "README.md"))
                   else:
                       ok = os.path.exists(resolved)
                   if not ok:
                       broken.append((os.path.relpath(fpath), m.group(1), os.path.relpath(resolved)))
       print(f"Checked {checked} links")
       if broken:
           for f, link, resolved in broken:
               print(f"BROKEN: {f}: [{link}] -> {resolved}")
           sys.exit(1)
       print("All links resolve")
       PY
       ```
